class Invalid(ValueError):
    pass


def require(ok, label):
    if not ok:
        raise Invalid(label)


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()


def digest(data):
    return hashlib.sha256(data).hexdigest()


def read_json(path):
    return json.loads(Path(path).read_text())


def safe_file(root, name):
    part = Path(name)
    require(not part.is_absolute() and bool(name) and part.as_posix() == name and
            all(not p.startswith(".") for p in part.parts), "nonliteral payload path")
    target = root / part
    current = target
    while current != root:
        require(not current.is_symlink(), "payload symlink")
        current = current.parent
    require(target.is_file(), "missing payload file")
    return target


def load_module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    require(Path(module.__file__).resolve() == path.resolve(), "module source path")
    return module

    expected_inputs = {CONFIG, arm_path, OWNER + "/protocol.txt"}
    expected_inputs.update(v["path"] for v in config["sources"].values())
    require(set(spec["inputs"]) == expected_inputs and len(spec["inputs"]) == len(expected_inputs), "exact inputs")
    require(set(spec["outputs"]) == set(CAPS) and spec["timeout_seconds"] == 360 and
            spec["depends_on"] == [], "manifest outputs and independent preparation")
    require(manifest["environment_id"] == config["environment"]["id"] and
            manifest["max_result_bytes"] == 16777216 and
            manifest["minimum_free_bytes"] >= 10737418240, "manifest environment and storage")
    require(manifest["spec_sha256"] == digest(canonical(spec)), "manifest spec digest")
    seen, total = set(), 0
    for row in manifest["files"]:
        require(row["path"] not in seen, "duplicate manifest entry")
        data = safe_file(bundle, row["path"]).read_bytes()
        require(len(data) == row["bytes"] and digest(data) == row["sha256"], "frozen payload bytes")
        seen.add(row["path"])
        total += len(data)
    require(len(seen) <= 128 and total <= 33554432, "full bundle bounds")
    require({"files/" + p for p in [spec["entrypoint"], *spec["inputs"]]} <= seen, "manifest contains explicit payload")
    inputs = {}
    for alias, row in config["sources"].items():
        p = safe_file(files, row["path"])
        data = p.read_bytes()
        require(len(data) == row["bytes"] and digest(data) == row["sha256"], "source bytes: " + alias)
        inputs[alias] = p
    for name, wanted in config["canonical_source_hashes"].items():
        require(digest(safe_file(files, name).read_bytes()) == wanted, "canonical source bytes")
    arm_config = read_json(safe_file(files, arm_path))
    validate_config(config, arm_config)
    require(spec["id"] == arm_config["study_id"] and spec["owner"] == arm_config["owner"], "manifest owner")
    return inputs, arm_config, {"bundle_id": digest(canonical(manifest)),
            "spec_sha256": manifest["spec_sha256"], "files": len(seen), "bytes": total}


class CountedRNG:
    """Pass the identical call through to PCG64; count requests, never draw extra."""
    def __init__(self, generator, dtype, shape):
        self.generator, self.dtype, self.shape = generator, dtype, shape
        self.calls = 0

    @property
    def bit_generator(self):
        return self.generator.bit_generator

    def standard_normal(self, shape, *, dtype):
        require(shape == self.shape and dtype == self.dtype, "noise draw schedule")
        self.calls += 1
        return self.generator.standard_normal(shape, dtype=dtype)


def evolve(state, step, finite, observe, total_steps, stride):
    """Orchestration only. Fixtures supply fake state/callbacks, never a simulator."""
    require(state["t_step"] == 0 and finite(state), "initial numerical validity")
    observe(state, 0)
    for index in range(1, total_steps + 1):
        step(state, 1)
        require(state["t_step"] == index, "one and only one native step")
        require(finite(state), "nonfinite or changed numerical state at step " + str(index))
        observe(state, index) if index % stride == 0 else None
    require(state["t_step"] == total_steps, "complete native horizon")


def write_bounded(output, name, data):
    require(len(data) <= CAPS[name], "output cap: " + name)
    with (output / name).open("wb") as stream:
        stream.write(data)
        stream.flush()


def write_summary(output, summary):
    write_bounded(output, "summary.json", canonical(summary) + b"\n")


def empty_archive():
    data = io.BytesIO()
    with zipfile.ZipFile(data, "w", compression=zipfile.ZIP_STORED):
        pass
    return data.getvalue()


def compact_observation(index, step, extrema, clips, errors, rng_hash):
    row = {"i": index, "s": step, "t": step / 50, "a": 250 + step / 50,
           "e": extrema, "c": clips, "q": [format(v, ".9g") for v in errors], "r": rng_hash}
    data = canonical(row) + b"\n"
    require(len(data) <= ROW_CAP, "observation row bound")
    return data


def save_archive(np, helper, output, arm, indices, baseline, detailed, context, times):
    count = len(times)
    arrays = {"readout_q8": np.asarray(detailed, dtype=np.uint8).reshape(count, 2, 4748),
              "context_q4": np.asarray(context, dtype=np.uint8).reshape(count, 128, 128),
              "readout_indices": indices, "baseline_readout_q8": baseline,
              "times": np.asarray(times, dtype="<f8"),
              "arm_code": np.frombuffer(arm.encode().ljust(2, b"\0"), dtype=np.uint8)}
    data = io.BytesIO()
    np.savez(data, **arrays)
    raw = data.getvalue()
    require(len(raw) <= CAPS["trajectory.npz"], "trajectory archive cap")
    schema = {k: (v.dtype.str, v.shape) for k, v in arrays.items()}
    write_bounded(output, "trajectory.npz", raw)
    decoded = helper.read_npz(np, output / "trajectory.npz", schema)
    require(all(decoded[k].tobytes() == arrays[k].tobytes() for k in arrays), "archive roundtrip")
    return {"path": "trajectory.npz", "bytes": len(raw), "sha256": digest(raw), "observations": count}

class BoundedLog:
    def __init__(self, stream, counter):
        self.stream, self.counter = stream, counter

    def write(self, text):
        size = len(text.encode("utf-8"))
        require(self.counter[0] + size <= 9000, "stdout/stderr Python log cap")
        self.counter[0] += size
        return self.stream.write(text)

    def flush(self):
        return self.stream.flush()


def actual_stdout_bytes(stream):
    try:
        return os.fstat(stream.fileno()).st_size
    except (AttributeError, OSError, io.UnsupportedOperation):
        return 0  # In-memory synthetic fixture only; worker stdout is a file.


def main():
    bundle = Path(os.environ["RESEARCH_BUNDLE_ROOT"]).resolve()
    files = bundle / "files"
    output = Path(os.environ["PHYSIM_SWARM_OUTPUT"]).resolve()
    require(files.is_dir() and output != bundle and bundle not in output.parents, "runner input/output directories")
    output.mkdir(parents=True, exist_ok=True)
    require(not any((output / name).exists() for name in CAPS), "output files already exist")
    for key in THREAD_KEYS:
        os.environ[key] = "1"
    require(os.environ.get("BLOBKIT_SKIP_LOCK") != "1", "lock bypass forbidden")
    write_summary(output, {"complete": False, "execution_valid": False, "disposition": "startup_incomplete", "no_retry": True})
    write_bounded(output, "trajectory.npz", empty_archive())
    write_bounded(output, "observations.jsonl", b"")
    started, real_stdout = time.monotonic(), sys.stdout
    counter = [0]
    try:
        with contextlib.redirect_stdout(BoundedLog(sys.stdout, counter)), contextlib.redirect_stderr(BoundedLog(sys.stderr, counter)):
            summary = worker_run(bundle, files, output)
        message = canonical({"study": summary["study_id"], "arm": summary["arm"], "complete": True}).decode() + "\n"
        real_stdout.flush()
        require(actual_stdout_bytes(real_stdout) + len(message.encode()) <= STDOUT_CAP, "actual stdout cap")
        real_stdout.write(message)
        real_stdout.flush()
        require(actual_stdout_bytes(real_stdout) <= STDOUT_CAP, "final stdout cap")
        summary["worker_wall_seconds"] = time.monotonic() - started
        summary["stdout_python_bytes"] = counter[0] + len(message.encode())
        write_summary(output, summary)  # Last write is the only complete-valid marker.
        return 0
    except Exception as exc:
        summary = read_json(output / "summary.json")
        summary.update(complete=False, execution_valid=False, disposition="invalid_or_incomplete", no_retry=True,
                       error_type=type(exc).__name__, reason=str(exc)[:500], worker_wall_seconds=time.monotonic()-started)
        write_summary(output, summary)
        return 2
